cmake_minimum_required(VERSION 3.25)
project(shadow_culling_evidence LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(Dawn CONFIG REQUIRED)
set(MP_SOURCE_ROOT "" CACHE PATH "MetroidPrimePort checkout containing the cascade filtering patch")
if(NOT EXISTS "${MP_SOURCE_ROOT}/extern/aurora/lib/gfx/shadow_cascade.cpp")
  message(FATAL_ERROR "Pass -DMP_SOURCE_ROOT=/path/to/MetroidPrimePort")
endif()
add_executable(shadow_culling_evidence gpu-test.cpp "${MP_SOURCE_ROOT}/extern/aurora/lib/gfx/shadow_cascade.cpp")
target_compile_features(shadow_culling_evidence PRIVATE cxx_std_20)
target_compile_options(shadow_culling_evidence PRIVATE -Wall -Wextra -Werror)
target_include_directories(shadow_culling_evidence PRIVATE "${MP_SOURCE_ROOT}/extern/aurora/lib")
target_link_libraries(shadow_culling_evidence PRIVATE dawn::webgpu_dawn)
